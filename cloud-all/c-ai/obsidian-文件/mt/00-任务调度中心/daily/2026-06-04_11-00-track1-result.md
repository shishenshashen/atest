## §1 M18 API 设计

### 公开 API (5 方法 + 辅助)

```mql5
class CCorrelationFilter {
public:
   // 1. 配置
   bool        Init(string &symbols[]);              // 加载品种列表, SymbolSelect
   void        SetDefaultDays(int days);             // 默认回看天数 (>=2)
   int         DefaultDays() const;                  // 查默认天数
   void        Clear();                              // 清空所有 buffer

   // 2. 加载
   int         LoadHistoricalCloses(string symbol, int days = 0);   // 拉日线 close, 返 bar 数

   // 3. 计算
   double      CalcCorr(string sym1, string sym2);   // Pearson 系数 -1~+1

   // 4. 对冲检查
   bool        IsHedgeExposed(string newSymbol, ulong magic, double threshold = 0.7);

   // 5. 调试
   string      DumpCorr();                           // 矩阵字符串

   // 辅助查询
   int         SymbolCount() const;
   string      LastError() const;
   string      GetSymbol(int idx) const;
   int         GetCloseCount(string symbol) const;
};
```

### 私有 buffer

```mql5
struct CorrSeries {
   string   symbol;       // 品种名 (trim 后)
   double   closes[];     // close 价, 倒序 ([0]=最新)
   int      count;        // bar 数
};

private:
   CorrSeries   _series[];      // 每个品种一个时间序列
   int          _nSymbols;      // 已 Init 的品种数
   int          _defaultDays;   // LoadHistoricalCloses 默认回看天数
   string       _lastError;     // 最近一次失败的错误信息

   int          _FindIndex(string symbol) const;       // 找索引
   string       _NormSymbol(string s) const;           // trim
   static double _Pearson(const double &x[], const double &y[], int n);
   static double _Mean(const double &v[], int n);
   static double _Sumsq(const double &v[], int n, double m);
```

### 关键算法 (Pearson)

```
r = sum((xi-mx)*(yi-my)) / sqrt(sum((xi-mx)^2) * sum((yi-my)^2))
范围 [-1, +1]; 0 = 无线性相关; ±1 = 完全线性
数据不足 (<2) / 零方差 (sxx=0 或 syy=0) → 返 0
数值夹紧到 [-1, +1] 防止浮点误差
```
