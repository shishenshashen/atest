## 画买卖箭头

```mql5
datetime t = iTime(_Symbol, _Period, 0);
double   p = iClose(_Symbol, _Period, 0);

// 多箭头（向上绿色）
string n1 = "MyEA_Buy_" + IntegerToString((int)t);
ObjectCreate(0, n1, OBJ_ARROW, 0, t, p);
ObjectSetInteger(0, n1, OBJPROP_ARROWCODE, 233);  // 向上
ObjectSetInteger(0, n1, OBJPROP_COLOR, clrLime);
ObjectSetInteger(0, n1, OBJPROP_WIDTH, 3);

// 空箭头（向下红色）
string n2 = "MyEA_Sell_" + IntegerToString((int)t);
ObjectCreate(0, n2, OBJ_ARROW, 0, t, p);
ObjectSetInteger(0, n2, OBJPROP_ARROWCODE, 234);  // 向下
ObjectSetInteger(0, n2, OBJPROP_COLOR, clrRed);
ObjectSetInteger(0, n2, OBJPROP_WIDTH, 3);

ChartRedraw(0);
```

## 画按钮（用户点击触发）

```mql5
// 创建
ObjectCreate(0, "Btn_Buy", OBJ_BUTTON, 0, 0, 0);
ObjectSetInteger(0, "Btn_Buy", OBJPROP_CORNER, CORNER_LEFT_UPPER);
ObjectSetInteger(0, "Btn_Buy", OBJPROP_XDISTANCE, 20);
ObjectSetInteger(0, "Btn_Buy", OBJPROP_YDISTANCE, 100);
ObjectSetInteger(0, "Btn_Buy", OBJPROP_XSIZE, 100);
ObjectSetInteger(0, "Btn_Buy", OBJPROP_YSIZE, 30);
ObjectSetString (0, "Btn_Buy", OBJPROP_TEXT, "BUY");
ObjectSetString (0, "Btn_Buy", OBJPROP_FONT, "Arial");
ObjectSetInteger(0, "Btn_Buy", OBJPROP_FONTSIZE, 12);
ObjectSetInteger(0, "Btn_Buy", OBJPROP_COLOR, clrWhite);
ObjectSetInteger(0, "Btn_Buy", OBJPROP_BGCOLOR, clrGreen);

// 处理点击
void OnChartEvent(const int id, const long &lparam,
                  const double &dparam, const string &sparam) {
   if (id == CHARTEVENT_OBJECT_CLICK) {
      if (sparam == "Btn_Buy") {
         // 开多
         ObjectSetInteger(0, "Btn_Buy", OBJPROP_STATE, false);
      }
   }
}
```

## 删所有本 EA 的对象

```mql5
void DeleteMyObjects(string prefix) {
   for (int i = ObjectsTotal(0) - 1; i >= 0; i--) {
      string n = ObjectName(0, i);
      if (StringFind(n, prefix) == 0) ObjectDelete(0, n);
   }
   ChartRedraw(0);
}
```

## 颜色常量速查
- `clrRed` `clrLime` `clrBlue` `clrYellow` `clrWhite` `clrBlack`
- `clrOrange` `clrMagenta` `clrCyan` `clrGray` `clrNONE`
- 自定义：`C'255,128,0'`

## 必看陷阱
- **加前缀**（`MyEA_`）避免冲突
- **OnDeinit 清理**
- 一次画超过 1000 个对象会卡
- 颜色要鲜艳，不然 K 线上看不清
- ChartRedraw() **不要在 OnTick 高频调**（影响性能）
